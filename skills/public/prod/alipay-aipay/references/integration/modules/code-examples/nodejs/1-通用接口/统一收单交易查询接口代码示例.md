# 统一收单交易查询接口 - Node.js 示例

## 请求示例

```javascript
// alipaySdk 必须由项目中当前已验证的生效配置工厂提供，禁止在源码中粘贴凭证。
async function queryTrade(alipaySdk, outTradeNo) {
  const result = await alipaySdk.exec("alipay.trade.query", {
    bizContent: {
      out_trade_no: outTradeNo,
    },
  });
  const response = normalizeTradeQueryResponse(result);

  if (response.code === "10000") {
    const tradeStatus = response.tradeStatus ?? response.trade_status ?? "UNKNOWN";
    return {
      outTradeNo: response.out_trade_no || outTradeNo,
      tradeStatus,
      result: response,
    };
  }

  // 非 10000 只表示本次查询失败，不能据此推断订单失败或沙箱数据已清理。
  return {
    outTradeNo,
    tradeStatus: "UNKNOWN",
    queryError: {
      code: response.code || "UNKNOWN",
      msg: response.msg || "",
      subCode: response.sub_code || "",
      subMsg: response.sub_msg || "",
    },
    result: response,
  };
}

function normalizeTradeQueryResponse(result) {
  return result?.alipay_trade_query_response || result || {};
}
```
