# 统一收单交易退款接口 - Node.js 示例

## 请求示例

```javascript
// alipaySdk 必须由项目中当前已验证的生效配置工厂提供，禁止在源码中粘贴凭证。
async function refundTrade(alipaySdk, request) {
  const result = await alipaySdk.exec("alipay.trade.refund", {
    bizContent: {
      out_trade_no: request.outTradeNo,
      refund_amount: request.refundAmount, // 十进制定点金额；累计退款不得超过原订单实付金额
      refund_reason: request.refundReason,
      out_request_no: request.outRequestNo, // 新退款使用新编号；同一次未知结果重试复用原编号和参数
    },
  });

  if (result.code !== "10000") {
    throw new Error("alipay.trade.refund failed");
  }
  return {
    result,
    confirmed: result.fund_change === "Y",
    nextAction: result.fund_change === "Y"
      ? "NONE"
      : "QUERY_WITH_SAME_OUT_REQUEST_NO",
  };
}
```
