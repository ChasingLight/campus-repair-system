# 统一收单下单并支付页面接口 - Node.js 示例

## 请求示例

```javascript
// ⚠️ 网站支付必须使用 pageExec() 方法，不能使用 exec()
// ⚠️ pageExec() 第二个参数必须传 'POST'
// ⚠️ notifyUrl 由服务端按当前 Coze 项目实际域名和真实 POST 路由构造；不要让前端或用户输入任意 notifyUrl
// ⚠️ 本地没有公网 HTTPS notifyUrl 时，直接省略 notifyUrl，不要向 SDK 传 undefined、空字符串或占位 URL
// ⚠️ returnUrl 默认传入，仅在用户明确关闭同步回跳时省略
// alipaySdk 必须由项目中当前已验证的生效配置工厂提供，禁止在源码中粘贴凭证。
function resolveCozeProjectOrigin(rawValue, sourceName = 'COZE_PROJECT_DOMAIN_DEFAULT') {
  if (typeof rawValue !== 'string' || !rawValue.trim()) {
    throw new Error(`${sourceName} is missing`);
  }
  const value = rawValue.trim();
  const withScheme = /^[a-z][a-z0-9+.-]*:\/\//i.test(value)
    ? value
    : `https://${value}`;
  const url = new URL(withScheme);
  if (url.protocol !== 'https:'
      || !url.hostname
      || url.username
      || url.password
      || (url.pathname !== '/' && url.pathname !== '')
      || url.search
      || url.hash) {
    throw new Error(`${sourceName} must be an HTTPS origin or hostname`);
  }
  return url.origin;
}

function createCozePaymentUrls(outTradeNo, options = {}) {
  // 默认使用 COZE_PROJECT_DOMAIN_DEFAULT；如用户明确指定自定义域名，
  // 先按 COZE_PROJECT_DOMAIN_ALL 核对它属于当前项目，再把该域名作为服务端可信配置传入。
  const hasSelectedDomain = typeof options.projectDomain === 'string' && Boolean(options.projectDomain.trim());
  const projectDomain = hasSelectedDomain
    ? options.projectDomain
    : process.env.COZE_PROJECT_DOMAIN_DEFAULT;
  const origin = resolveCozeProjectOrigin(
    projectDomain,
    hasSelectedDomain ? 'selected Coze project domain' : 'COZE_PROJECT_DOMAIN_DEFAULT'
  );
  const returnUrl = new URL('/pay/result', `${origin}/`);
  returnUrl.searchParams.set('outTradeNo', outTradeNo);
  const notifyUrl = new URL('/api/pay/notify', `${origin}/`);
  return {
    returnUrl: returnUrl.toString(),
    notifyUrl: notifyUrl.toString(),
  };
}

async function createWebPaymentHtml(alipaySdk, request, runtimeOptions = {}) {
  const needsReturnUrl = !runtimeOptions.disableReturnUrl;
  const needsNotifyUrl = runtimeOptions.hasPublicNotifyUrl !== false;
  const paymentUrls = needsReturnUrl || needsNotifyUrl
    ? createCozePaymentUrls(request.outTradeNo, runtimeOptions)
    : {};
  const requestOptions = {
    bizContent: {
      out_trade_no: request.outTradeNo,
      total_amount: request.totalAmount,
      subject: request.subject,
      product_code: "FAST_INSTANT_TRADE_PAY",
    },
  };
  if (needsReturnUrl) requestOptions.returnUrl = paymentUrls.returnUrl;
  if (needsNotifyUrl) requestOptions.notifyUrl = paymentUrls.notifyUrl;

  const paymentHtml = await alipaySdk.pageExec(
    "alipay.trade.page.pay",
    "POST",
    requestOptions
  );
  return paymentHtml;
}
```

### 前端在当前页面提交支付表单

```javascript
// pageExec 返回的是包含 <form> 和脚本的 HTML 字符串。
// innerHTML 不会执行其中的 <script>；应提取 form 后显式提交。
// 支付按钮在请求期间显示应用内 loading 并禁用；取得响应后直接调用本函数。
// showInlineError 由项目现有 UI 提供，禁止改用 window.alert/confirm/prompt。
function submitPaymentForm(data, expectedGatewayEndpoint, showInlineError) {
  const fail = () => {
    showInlineError('暂时无法发起支付，请稍后重试。');
    return false;
  };
  if (!data.paymentHtml) {
    return fail();
  }

  const template = document.createElement('template');
  template.innerHTML = data.paymentHtml.trim();
  const form = template.content.querySelector('form');
  if (!form) {
    return fail();
  }

  let action;
  try {
    action = new URL(form.action, window.location.href);
  } catch {
    return fail();
  }
  const actionEndpoint = `${action.origin}${action.pathname}`;
  if (action.protocol !== 'https:'
      || actionEndpoint !== expectedGatewayEndpoint
      || form.method.toLowerCase() !== 'post') {
    return fail();
  }

  form.removeAttribute('target');
  form.hidden = true;
  document.body.appendChild(form);
  HTMLFormElement.prototype.submit.call(form);
  return true;
}

// 沙箱开发阶段传入固定值：
// 'https://openapi-sandbox.dl.alipaydev.com/gateway.do'
// 后续独立生产技能替换配置时由生产配置提供：
// 'https://openapi.alipay.com/gateway.do'
// 当前 Skill 不在本示例中实现生产转换；两者不得同时保留为自动切换或失败回退分支。

// 默认不增加第二次确认。用户明确要求或项目既定体验包含应用内确认 modal 时，
// 才从该 modal 的唯一确认回调调用 submitPaymentForm；不要使用浏览器原生弹窗。
// prototype.submit.call 避免名为 submit 的控件遮蔽方法，不是因为 React 会拦截原生提交。
// template 中的 form 跨域不等于浏览器会拒绝提交；平台阻断必须以实际网络/控制台证据判断。

// 错误写法：
// window.location.href = data.paymentHtml; // HTML 表单不是 URL
// window.location.href = form.action;      // 会丢失 POST hidden input
// action.searchParams.set(input.name, input.value); // 禁止把 POST form 改造成 GET URL
// container.innerHTML = data.paymentHtml;  // 不能依赖其中的脚本自动执行
// document.write(data.paymentHtml);        // 会破坏当前文档和应用状态
// window.open(action);                     // 会打开新标签页，Coze 预览禁止
// form.target = '_blank';                  // 会改变为新窗口/标签页
// window.confirm('确认支付');               // Coze 支付流程禁止浏览器原生对话框
// `https://${process.env.COZE_PROJECT_DOMAIN_DEFAULT}`; // 环境值可能已含 scheme
```
